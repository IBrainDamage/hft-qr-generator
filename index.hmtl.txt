    <script>
        function generateQR() {
            const store = document.getElementById('storeInput').value;
            const username = document.getElementById('userInput').value;
            const password = document.getElementById('passInput').value;
            
            const placeholder = document.getElementById('placeholder');
            const qrBox = document.getElementById('qrBox');
            const canvas = document.getElementById('qrCanvas');
            const downloadBtn = document.getElementById('downloadBtn');

            if (!store && !username && !password) {
                placeholder.style.display = 'block';
                qrBox.style.display = 'none';
                downloadBtn.disabled = true;
                return;
            }

            // Build tab-delimited payload
            const payload = store + '\t' + username + '\t' + password;

            try {
                canvas.width = 280;
                canvas.height = 280;

                // 1. Generate base high error-correction QR code
                const qr = new QRious({
                    element: canvas,
                    value: payload,
                    size: 280,
                    level: 'H',
                    foreground: '#0f172a',
                    background: '#ffffff'
                });

                const ctx = canvas.getContext('2d', { willReadFrequently: true });
                let imgData = ctx.getImageData(0, 0, 280, 280);
                let pixels = imgData.data;

                // 2. Create offscreen mask for "HFT" letters
                let maskCanvas = document.createElement('canvas');
                maskCanvas.width = 280;
                maskCanvas.height = 280;
                let mCtx = maskCanvas.getContext('2d');
                
                mCtx.fillStyle = '#000000';
                mCtx.fillRect(0, 0, 280, 280);
                
                mCtx.fillStyle = '#ffffff';
                mCtx.font = '900 95px sans-serif';
                mCtx.textAlign = 'center';
                mCtx.textBaseline = 'middle';
                mCtx.fillText('HFT', 280 / 2, 280 / 2);

                let maskData = mCtx.getImageData(0, 0, 280, 280).data;

                // 3. Carve pixels into HFT pattern
                for (let i = 0; i < pixels.length; i += 4) {
                    if (maskData[i] > 128) {
                        let x = (i / 4) % 280;
                        let y = Math.floor((i / 4) / 280);
                        
                        if ((x % 6 < 4) && (y % 6 < 4)) {
                            pixels[i] = 15;     // R
                            pixels[i+1] = 23;   // G
                            pixels[i+2] = 42;   // B
                        } else {
                            pixels[i] = 255;
                            pixels[i+1] = 255;
                            pixels[i+2] = 255;
                        }
                    }
                }

                ctx.putImageData(imgData, 0, 0);

                placeholder.style.display = 'none';
                qrBox.style.display = 'inline-block';
                downloadBtn.disabled = false;
            } catch (err) {
                console.error("QR Carving error:", err);
            }
        }

        function clearForm() {
            document.getElementById('storeInput').value = '';
            document.getElementById('userInput').value = '';
            document.getElementById('passInput').value = '';
            generateQR();
        }

        function downloadQR() {
            const canvas = document.getElementById('qrCanvas');
            const a = document.createElement('a');
            a.href = canvas.toDataURL('image/png');
            a.download = 'HFT_3Subject_Carved_QR.png';
            a.click();
        }

        // Attach live listeners for instant matrix carving as you type
        ['storeInput', 'userInput', 'passInput'].forEach(id => {
            document.getElementById(id).addEventListener('input', generateQR);
        });
